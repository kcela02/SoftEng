D.2.1 Main Application (app.py)

# app.py - Application Factory Pattern
from flask import Flask, jsonify, redirect, url_for
from flask_cors import CORS
from flask_login import LoginManager
from flask_wtf.csrf import CSRFProtect
from flask_limiter import Limiter
from flask_limiter.util import get_remote_address
from flask_jwt_extended import JWTManager
from werkzeug.middleware.proxy_fix import ProxyFix
from config import config
from models import db, User
import os

def create_app(config_name=None):
    app = Flask(__name__)
    app.wsgi_app = ProxyFix(app.wsgi_app, x_for=1, x_proto=1, x_host=1)
    if config_name is None:
        config_name = os.getenv('FLASK_ENV', 'development')
    app.config.from_object(config[config_name])
    db.init_app(app)
    JWTManager(app)
    csrf = CSRFProtect(app)
    limiter = Limiter(
        app=app,
        key_func=get_remote_address,
        default_limits=app.config.get('RATELIMIT_DEFAULTS', ["200 per day", "50 per hour"]),
        storage_uri=app.config.get('RATELIMIT_STORAGE_URI', 'memory://'),
    )
    app.limiter = limiter
    CORS(app, origins=app.config.get('CORS_ORIGINS', []), supports_credentials=True)
    login_manager = LoginManager()
    login_manager.init_app(app)
    login_manager.login_view = 'main.index'
    @login_manager.user_loader
    def load_user(user_id):
        return db.session.get(User, int(user_id))
    from blueprints.auth import auth_bp
    from blueprints.main import main_bp
    from blueprints.api import api_bp
    from blueprints.mobile import mobile_bp
    app.register_blueprint(auth_bp)
    app.register_blueprint(main_bp)
    app.register_blueprint(api_bp)
    app.register_blueprint(mobile_bp)
    return app


D.2.2 Database Models (models/__init__.py)

# Key models — User, Product, Sale, Forecast, Alert
from flask_sqlalchemy import SQLAlchemy
from flask_login import UserMixin
from werkzeug.security import generate_password_hash, check_password_hash
from datetime import datetime

db = SQLAlchemy()

class User(UserMixin, db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(150), unique=True, nullable=False)
    email = db.Column(db.String(150), unique=True, nullable=False)
    password_hash = db.Column(db.String(255), nullable=False)
    role = db.Column(db.String(50), default='user')  # 'admin' or 'user'
    failed_login_attempts = db.Column(db.Integer, default=0)
    account_locked_until = db.Column(db.DateTime, nullable=True)
    def set_password(self, password):
        self.password_hash = generate_password_hash(password)
    def check_password(self, password):
        return check_password_hash(self.password_hash, password)

class Product(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(255), nullable=False)
    category = db.Column(db.String(100))
    unit_cost = db.Column(db.Float)
    current_stock = db.Column(db.Integer, default=0)

class Sale(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    product_id = db.Column(db.Integer, db.ForeignKey('product.id'), nullable=False)
    quantity = db.Column(db.Integer, nullable=False)
    price = db.Column(db.Float, nullable=False)
    sale_date = db.Column(db.DateTime, default=datetime.utcnow, index=True)

class Forecast(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    product_id = db.Column(db.Integer, db.ForeignKey('product.id'), nullable=False)
    forecast_date = db.Column(db.DateTime, nullable=False)
    predicted_quantity = db.Column(db.Integer)
    model_used = db.Column(db.String(50))  # 'ARIMA', 'LINEAR_REGRESSION'
    confidence_lower = db.Column(db.Float)
    confidence_upper = db.Column(db.Float)
    mae = db.Column(db.Float)
    rmse = db.Column(db.Float)

class Alert(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    product_id = db.Column(db.Integer, db.ForeignKey('product.id'), nullable=False)
    alert_type = db.Column(db.String(50))
    severity = db.Column(db.String(20))  # 'CRITICAL', 'HIGH', 'MEDIUM'
    message = db.Column(db.Text)
    recommended_order_qty = db.Column(db.Integer)
    is_active = db.Column(db.Boolean, default=True)
    is_acknowledged = db.Column(db.Boolean, default=False)

# ARIMA forecasting with automatic order selection (AIC) and confidence intervals
from statsmodels.tsa.arima.model import ARIMA
from models.utils import load_sales_data, create_future_dates
import numpy as np

def forecast_arima(db_conn, product_id, days_ahead=7, order=(5,1,0)):
    ts = load_sales_data(db_conn, product_id)
    if ts.empty or len(ts) < 30:
        return {"error": "Insufficient data (minimum 30 days required)."}
    best_aic = float('inf')
    best_model_fit = None
    for candidate_order in [(1,1,1), (2,1,2), (5,1,0), (2,0,2)]:
        try:
            model_fit = ARIMA(ts['y'], order=candidate_order).fit()
            if model_fit.aic < best_aic:
                best_aic = model_fit.aic
                best_model_fit = model_fit
        except Exception:
            continue
    forecast_result = best_model_fit.get_forecast(steps=days_ahead, alpha=0.2)
    predictions = forecast_result.predicted_mean
    conf_int = forecast_result.conf_int()
    future_dates = create_future_dates(ts, days_ahead)
    return [{'date': d, 'prediction': max(0, round(predictions.iloc[i])),
             'confidence_lower': max(0, round(conf_int.iloc[i, 0])),
             'confidence_upper': max(0, round(conf_int.iloc[i, 1]))
             } for i, d in enumerate(future_dates)]

D.3.1 Templates (Jinja2 / HTML)
---------------------------------
The system uses server-side rendered HTML templates (Jinja2). The base_dashboard.html
template includes Chart.js (CDN), Font Awesome icons, and the main CSS stylesheet.
All protected pages extend base_dashboard.html to inherit the navigation
bar, session-aware header, and static asset caching.

Key templates:
- templates/index.html         Login / landing page
- templates/admin.html         Main dashboard with metric cards, Forecast
                               Trend Analysis chart, Historical Performance chart,
                               and Top Products Sales Ranking chart
- templates/forecasting.html   Per-product ARIMA and Linear Regression
                               7/30-day forecast charts with confidence bands
- templates/products.html      Product CRUD management table
- templates/reports.html       Sales and inventory reports
- templates/settings.html      User and system settings


D.3.2 Frontend JavaScript (static/app.js)
------------------------------------------
Key JavaScript functions:
- getCsrfToken()         Reads the CSRF token from the meta tag for all API calls
- fetchWithCsrf(url, options)  Wrapper around fetch() that auto-injects the CSRF header
- vcCountUp(el, target)  Animates metric card numbers counting up on page load
- vcSkeleton(el, fetchFn) Shows a shimmer skeleton while data is loading

Chart.js is used for:
- Forecast Trend Analysis (line chart, canvas id="trendChart")
- Historical Performance (bar/line toggle, canvas id="monthlyChart")
- Sales Ranking - Top Products (bar chart, canvas id="topProductsChart")

This appendix provides instructions on how to use the Predictive Sales &

E.1 SYSTEM REQUIREMENTS
--------------------------------------------------------------------------------

Hardware Requirements:
- Computer/Laptop with internet connection
- Minimum 4GB RAM
- 100MB free storage

Software Requirements:
- Web Browser (Chrome, Firefox, Edge, or Safari - latest version)
- Python 3.x (for local deployment)
- Node.js (for local deployment)


E.2 INSTALLATION INSTRUCTIONS
--------------------------------------------------------------------------------

For Local Deployment:

1. Clone the repository
   git clone [REPOSITORY_URL]
   cd "my app"

2. Create and activate a virtual environment
   python -m venv .venv
   .venv\Scripts\activate         (Windows)
   source .venv/bin/activate      (macOS / Linux)

3. Install Python dependencies
   pip install -r requirements.txt

4. Set environment variables (create a .env file):
   SECRET_KEY=your-secret-key
   DATABASE_URL=postgresql://user:password@host/dbname
   FLASK_ENV=development

5. Run the application
   python app.py

6. Access the application
   Open browser and navigate to: http://127.0.0.1:5000

For Cloud Deployment (Render):
   Push code to the connected GitHub repository. Render automatically
   builds and deploys using the configuration in render.yaml.


E.3 USER GUIDE
--------------------------------------------------------------------------------

E.3.1 Logging In
-----------------
1. Open the application in a web browser
2. Enter your username and password
3. Click "Login" button
4. You will be redirected to the dashboard

Default Admin Credentials:
- Username: admin
- Password: admin123

  NOTE: The system forces a password change on the first login.
  Please change the default password immediately after first access.


E.3.2 Uploading Sales Data
---------------------------
1. Click on "Upload" or "Data Upload" in the navigation menu
2. Click "Choose File" button
3. Select your CSV file from your computer
4. Click "Upload" button
5. Wait for processing to complete
6. View the validation results

CSV File Format Requirements:
- Required columns: date, product_name, quantity, price
- Date format: YYYY-MM-DD
- File format: .csv only


E.3.3 Viewing Forecasts
------------------------
1. Click on "Forecasts" in the navigation menu
2. Select a product from the dropdown
3. View the forecast chart and predicted values
4. Use date filters to adjust the time range


E.3.4 Managing Inventory
-------------------------
1. Click on "Inventory" in the navigation menu
2. View current stock levels
3. To add a new product:
   - Click "Add Product"
   - Fill in product details
   - Set reorder level
   - Click "Save"
4. To update stock:
   - Click on the product
   - Update quantity
   - Click "Update"


E.3.5 Viewing and Managing Alerts
----------------------------------
1. Click on "Alerts" in the navigation menu
2. View all pending restock alerts
3. To resolve an alert:
   - Click on the alert
   - Update inventory (add stock)
   - Mark alert as "Resolved"


E.3.6 Logging Out
------------------
1. Click on your username/profile in the top right corner
2. Click "Logout" button
3. You will be redirected to the login page


E.4 TROUBLESHOOTING
--------------------------------------------------------------------------------

| Problem | Solution |
|---------|----------|
| Cannot login | Check credentials, contact admin |
| CSV upload fails | Verify file format and columns |
| Forecast not generating | Ensure sufficient historical data |
| Page not loading | Clear browser cache, refresh |
| System slow | Check internet connection |


E.5 FREQUENTLY ASKED QUESTIONS (FAQs)
--------------------------------------------------------------------------------

Q: What if my CSV file has errors?
A: The system will show validation errors. Fix the issues and re-upload.

Q: How much historical data do I need?
A: For accurate forecasts, at least 12 months of data is recommended.

Q: Can I export the reports?
A: Yes. Use the Export Report button on the Reports page to download a CSV of sales data and alerts.
